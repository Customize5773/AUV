cmake_minimum_required(VERSION 3.16)
project(sauvc_bridge)
# Match the existing Fortress build fallback on this Ubuntu 24.04 host.
set(local_protobuf "${CMAKE_CURRENT_SOURCE_DIR}/../../.native-deps/protobuf-3.12/usr/include")
if(EXISTS "${local_protobuf}/google/protobuf/generated_message_table_driven.h")
  find_library(local_protobuf_library NAMES libprotobuf.so.23 REQUIRED)
  set(Protobuf_INCLUDE_DIR "${local_protobuf}")
  set(Protobuf_LIBRARY_RELEASE "${local_protobuf_library}")
  set(Protobuf_LIBRARY "${local_protobuf_library}")
endif()
find_package(ament_cmake REQUIRED)
find_package(ros_gz_bridge REQUIRED)
if(EXISTS "${local_protobuf}/google/protobuf/generated_message_table_driven.h")
  find_library(local_yaml_cpp_library NAMES libyaml-cpp.so.0.7 REQUIRED)
  get_target_property(bridge_libraries ros_gz_bridge::ros_gz_bridge INTERFACE_LINK_LIBRARIES)
  list(REMOVE_ITEM bridge_libraries yaml-cpp)
  list(APPEND bridge_libraries "${local_yaml_cpp_library}")
  set_property(TARGET ros_gz_bridge::ros_gz_bridge PROPERTY INTERFACE_LINK_LIBRARIES "${bridge_libraries}")
endif()
add_executable(parameter_bridge src/parameter_bridge.cc)
target_compile_features(parameter_bridge PRIVATE cxx_std_17)
target_link_libraries(parameter_bridge ros_gz_bridge::ros_gz_bridge)
install(TARGETS parameter_bridge DESTINATION lib/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  add_executable(bridge_lifetime_test test/lifetime.cc)
  target_compile_features(bridge_lifetime_test PRIVATE cxx_std_17)
  target_link_libraries(bridge_lifetime_test ros_gz_bridge::ros_gz_bridge)
  add_test(NAME bridge_lifetime COMMAND bridge_lifetime_test)
  set_tests_properties(bridge_lifetime PROPERTIES TIMEOUT 30
    ENVIRONMENT "ROS_DOMAIN_ID=231;IGN_PARTITION=sauvc_bridge_lifetime")
endif()
ament_package()
